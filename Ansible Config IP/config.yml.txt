---
- name: Configure VM IPs and reboot
  hosts: vms
  become: true
  gather_facts: true

  tasks:
    - name: Set static IP address using nmcli
      ansible.builtin.shell:
        cmd: >
          nmcli con mod 'Connexion filaire 1'
          ipv4.addresses {{ host_ip }}/24
          ipv4.gateway {{ gateway }}
          ipv4.method manual
      when: ansible_default_ipv4.interface == 'ens160'
      register: ip_change
      changed_when: "'successfully' in ip_change.stdout"

    - name: Debug output after IP change
      ansible.builtin.debug:
        msg: "IP change task result: {{ ip_change.stdout }}"
      when: ip_change is changed

    - name: Check conditions for reboot
      ansible.builtin.debug:
        msg: "Interface: {{ ansible_default_ipv4.interface }}, IP changed: {{ ip_change is changed }}"

    - name: Reboot the system if IP was changed
      ansible.builtin.reboot:
        msg: "Rebooting system for IP changes to take effect"
        connect_timeout: 5
        reboot_timeout: 300
        pre_reboot_delay: 0
        post_reboot_delay: 30
        test_command: uptime
      when:
        - ansible_default_ipv4.interface == 'ens160'
        - ip_change is changed
      register: reboot_result

    - name: Debug output after reboot
      ansible.builtin.debug:
        msg: "System has been rebooted successfully."
      when: reboot_result.rebooted | default(false)
